cmake_minimum_required(VERSION 3.20)
project(CodexQuotaStrip VERSION 3.0.0 LANGUAGES CXX)
if(WIN32)
  message(FATAL_ERROR "Use Build.ps1 for the existing Windows application.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(native-core native/NativeCore.cpp)
target_link_libraries(native-core PUBLIC Threads::Threads)
target_compile_options(native-core PRIVATE -Wall -Wextra)
add_executable(core-tests native/CoreTests.cpp)
add_executable(fake-server native/FakeServer.cpp)
add_executable(query-tests native/QueryTests.cpp)
add_executable(runtime-tests native/RuntimeTests.cpp)
add_executable(proxy-config-tests source/ProxyConfigTests.cpp)
target_link_libraries(query-tests PRIVATE native-core)
target_link_libraries(runtime-tests PRIVATE native-core)
enable_testing()
add_test(NAME core COMMAND core-tests)
add_test(NAME query COMMAND query-tests $<TARGET_FILE:fake-server>)
add_test(NAME runtime COMMAND runtime-tests $<TARGET_FILE:fake-server>)
add_test(NAME proxy-config COMMAND proxy-config-tests)
set_tests_properties(core query runtime proxy-config PROPERTIES TIMEOUT 15)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  add_executable(CodexQuotaStrip MACOSX_BUNDLE native/MacApp.mm)
  target_compile_options(CodexQuotaStrip PRIVATE -fobjc-arc)
  target_link_libraries(CodexQuotaStrip PRIVATE native-core "-framework AppKit" "-framework ApplicationServices")
  set_target_properties(CodexQuotaStrip PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/native/Info.plist.in")
  # 预览包使用 ad-hoc 完整性签名；没有 Apple Developer ID，也没有公证。
  add_custom_command(TARGET CodexQuotaStrip POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:CodexQuotaStrip>/Contents/Resources"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "$<TARGET_BUNDLE_DIR:CodexQuotaStrip>/Contents/Resources/LICENSE"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.md" "$<TARGET_BUNDLE_DIR:CodexQuotaStrip>/Contents/Resources/THIRD_PARTY_NOTICES.md"
    COMMAND /usr/bin/codesign --force --sign - --identifier io.github.jyongli.codex-quota-strip "$<TARGET_BUNDLE_DIR:CodexQuotaStrip>")
  install(TARGETS CodexQuotaStrip BUNDLE DESTINATION .)
  install(FILES LICENSE DESTINATION CodexQuotaStrip.app/Contents/Resources)
  install(FILES THIRD_PARTY_NOTICES.md DESTINATION CodexQuotaStrip.app/Contents/Resources)
  set(CPACK_GENERATOR ZIP)
  set(CPACK_PACKAGE_FILE_NAME "CodexQuotaStrip-macos-universal")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
  pkg_check_modules(AYATANA REQUIRED IMPORTED_TARGET ayatana-appindicator3-0.1)
  find_package(X11 REQUIRED)
  add_executable(CodexQuotaStrip native/LinuxApp.cpp)
  target_link_libraries(CodexQuotaStrip PRIVATE native-core PkgConfig::GTK PkgConfig::AYATANA X11::X11)
  install(TARGETS CodexQuotaStrip RUNTIME DESTINATION bin)
  install(FILES native/codex-quota-strip.desktop DESTINATION share/applications)
  set(CPACK_GENERATOR "DEB;TGZ")
  set(CPACK_PACKAGE_FILE_NAME "CodexQuotaStrip-linux-x86_64")
  set(CPACK_DEBIAN_PACKAGE_NAME codex-quota-strip)
  set(CPACK_DEBIAN_PACKAGE_VERSION "3.0.0~preview4")
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE amd64)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "libgtk-3-0, libayatana-appindicator3-1, libx11-6, libstdc++6, libc6 (>= 2.35)")
  set(CPACK_DEBIAN_PACKAGE_MAINTAINER "jyongli <49582085+jyongli@users.noreply.github.com>")
  set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
endif()
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md README.en.md DESTINATION share/doc/codex-quota-strip)
set(CPACK_PACKAGE_NAME codex-quota-strip)
set(CPACK_PACKAGE_VERSION "3.0.0")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Codex quota tray and window-following display")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
include(CPack)
